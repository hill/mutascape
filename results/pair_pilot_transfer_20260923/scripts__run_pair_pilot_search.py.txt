"""Compare search-method rankings on measured and pilot-generated landscapes."""

import argparse
import json
import time

import numpy as np
import pandas as pd
from joblib import Parallel, delayed

from mutascape.benchmarks.library import SearchSpace
from mutascape.benchmarks.methods import METHODS, make_features, run_method
from mutascape.benchmarks.topology import supported_cube
from scripts.test_pair_pilot_transfer import DRAW_SEEDS, OUT, PROTEINS, digest, load_protein


def run_search(name, split, budget, model, draw):
    directory = OUT / name / f"p48_s{split}"
    result = directory / f"search_b{budget}_{model}_{draw}.csv"
    if result.exists():
        return
    dataset, _ = load_protein(name)
    frame = dataset.variants.reset_index(drop=True)
    singles = frame.loc[frame.depth <= 1]
    cube = supported_cube(dataset, singles)
    shape = tuple(len(cube.alphabets[p]) for p in cube.positions)
    codes = np.array(
        [
            [cube.alphabets[p].index(m.get(p, cube.wild_type[p])) for p in cube.positions]
            for m in frame.mutations
        ]
    )
    flat_indices = np.ravel_multi_index(codes.T, shape)
    pilot_ids = pd.read_csv(OUT / name / f"pilot_{split}.csv").row.to_numpy()[:48]
    included = (frame.depth > 1).to_numpy()
    included[pilot_ids] = False
    mask = np.zeros(len(frame), dtype=bool)
    mask[flat_indices[included]] = True
    space = SearchSpace.from_cube(cube, mask.reshape(shape))
    values = (
        frame.fitness.to_numpy()
        if model == "measured"
        else np.load(directory / f"{model}_{draw}.npy")
    )
    flat_values = np.empty(len(frame))
    flat_values[flat_indices] = values
    search_values = flat_values[space.flat_rows]
    features = make_features(space)
    rows = []
    for method in METHODS:
        for trial in range(10):
            started = time.perf_counter()
            best = run_method(
                method,
                space,
                search_values,
                features,
                budget=budget,
                seed=2026092600 + split * 1000 + trial * 137,
            )
            regret = (search_values.max() - best) / np.ptp(search_values)
            rows.append(
                dict(
                    protein=name,
                    split=split,
                    budget=budget,
                    model=model,
                    draw=draw,
                    method=method,
                    trial=trial,
                    best=best,
                    regret=regret,
                    space_size=space.size,
                    seconds=time.perf_counter() - started,
                )
            )
    pd.DataFrame(rows).to_csv(result, index=False)
    print(f"Searched {name} split {split} budget {budget} {model} {draw}", flush=True)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--jobs", type=int, default=4)
    args = parser.parse_args()
    protocol = json.loads((OUT / "protocol.json").read_text())
    for path, checksum in protocol["code_sha256"].items():
        assert digest(path) == checksum, path
    tasks = []
    for name in PROTEINS:
        for split in range(3):
            if not (OUT / name / f"p48_s{split}" / "done.json").exists():
                continue
            for budget in (48, 96):
                tasks.append((name, split, budget, "measured", 0))
                tasks.extend(
                    (name, split, budget, model, seed)
                    for model in ("pilot_reference", "learned_pairs")
                    for seed in DRAW_SEEDS
                )
    Parallel(n_jobs=args.jobs)(delayed(run_search)(*task) for task in tasks)
    print(f"Completed {len(tasks)} search groups", flush=True)


if __name__ == "__main__":
    main()
